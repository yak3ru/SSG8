"""Offscreen integration checks for the Phase 6 paranoia demonstration."""

from __future__ import annotations

import pytest
from PySide6.QtWidgets import QApplication

from ssg8.core.context import build_context
from ssg8.core.models import EventCategory
from ssg8.security.policies import DEFAULT_FILE_SIZE_MB
from ssg8.services.paranoia.states import ParanoiaState
from ssg8.storage.preferences import Preferences
from ssg8.storage.protection_store import ProtectionStore
from ssg8.ui import theme
from ssg8.ui.main_window import MainWindow


@pytest.fixture(scope="module")
def qapp():
    app = QApplication.instance() or QApplication([])
    theme.apply_theme(app)
    return app


@pytest.fixture
def window(qapp, tmp_path):
    context = build_context(
        Preferences(tmp_path / "preferences.json"),
        ProtectionStore(tmp_path / "data"),
        demo_root=tmp_path / "demo",
    )
    window = MainWindow(context)
    window.show()
    qapp.processEvents()
    yield window
    context.demo.cleanup()
    window.close()
    qapp.processEvents()


def test_phase6_pages_are_registered(window) -> None:
    assert {"paranoia", "demo", "recovery"} <= set(window.pages())
    for page_id in ("paranoia", "demo", "recovery"):
        window.navigate(page_id)
        assert window.current_page_id == page_id


def test_dashboard_updates_immediately_on_paranoia_change(window) -> None:
    window.context.paranoia.record_activity("browser_open", points=20)
    window.navigate("dashboard")
    QApplication.processEvents()

    dashboard = window.page("dashboard")
    assert dashboard._paranoia_card.value() == "20/100"
    assert dashboard._paranoia_state_card.value() == "Suspicious"
    assert dashboard._paranoia_chip.text() == "SUSPICIOUS"
    assert dashboard._paranoia_meter.score() == 20
    assert dashboard._paranoia_events_table.rowCount() >= 1
    assert dashboard._paranoia_transition_table.rowCount() >= 1


def test_controlled_browser_uses_website_engine(window) -> None:
    page = window.page("demo")
    window.navigate("demo")

    result = page.open_website("https://unfamiliar-demo.test/login")

    assert result.action == "allow"
    assert result.decision is not None
    assert result.decision.module == "Website Protection"
    assert window.context.paranoia.score() == 8
    assert page.visible_results() == 1
    assert any(
        entry.action == "allow" and entry.module_id == "website"
        for entry in window.context.events.entries()
    )


def test_controlled_download_uses_download_engine(window) -> None:
    page = window.page("demo")
    result = page.request_download("installer.exe")

    assert result.action == "block"
    assert result.decision is not None
    assert result.decision.module == "Download Protection"
    assert any(
        decision.module == "Download Protection"
        and decision.action.value == "block"
        for decision in window.context.protection.decisions()
    )


def test_launcher_follows_the_paranoia_profile(window) -> None:
    page = window.page("demo")

    normal = page.launch_application("calculator")
    assert normal.action == "allow"

    window.context.paranoia.advance_to_maximum_paranoia()
    blocked = page.launch_application("calculator")
    assert blocked.action == "block"
    assert "unauthorized mathematical" in blocked.warning.casefold()


def test_notification_control_receives_demo_notifications(window) -> None:
    window.context.state.set_notification_control_enabled(True)
    page = window.page("demo")

    result = page.context.demo.send_notification("general", "Demo message")

    assert result.decision is not None
    assert result.decision.module == "Notification Control"


def test_security_logs_cover_the_full_progression(window) -> None:
    context = window.context
    context.paranoia.advance_to_maximum_paranoia()
    context.demo.launch_application("notepad")
    report, _message = context.paranoia.run_self_audit()
    assert report is not None and report.has_violations
    context.paranoia.begin_quarantine()
    context.paranoia.complete_recovery()

    categories = {entry.category for entry in context.events.entries()}
    assert EventCategory.PARANOIA in categories
    assert EventCategory.SELF_AUDIT in categories
    assert EventCategory.RECOVERY in categories
    assert context.paranoia.state() is ParanoiaState.RECOVERY


def test_paranoia_page_shows_audit_and_quarantine(window) -> None:
    context = window.context
    window.navigate("paranoia")
    page = window.page("paranoia")

    context.paranoia.advance_to_maximum_paranoia()
    context.demo.launch_application("paint")
    QApplication.processEvents()
    assert page.state_text() == "MAXIMUM PARANOIA"

    page.run_audit()
    QApplication.processEvents()

    assert page.audit_rows() >= 6
    assert context.paranoia.state() is ParanoiaState.SELF_THREAT_DETECTED
    assert page.visible_activities() >= 1
    assert page.visible_transitions() >= 1

    context.paranoia.begin_quarantine()
    page.complete_recovery()
    QApplication.processEvents()
    assert page.state_text() == "RECOVERY"
    assert context.paranoia.score() == 0


def test_recovery_center_quick_and_full_reset(window) -> None:
    context = window.context
    window.navigate("recovery")
    page = window.page("recovery")

    context.paranoia.record_activity("browser_open", points=45)
    QApplication.processEvents()
    assert page.state_text() == "RESTRICTIVE"

    page.quick_reset()
    QApplication.processEvents()
    assert page.state_text() == "NORMAL"
    assert context.paranoia.score() == 0

    context.protection.update_settings(max_file_size_mb=12)
    page.perform_full_reset()
    QApplication.processEvents()

    assert context.protection.settings().max_file_size_mb == DEFAULT_FILE_SIZE_MB
    assert all(context.protection.module_states().values())
    assert context.paranoia.state() is ParanoiaState.NORMAL


def test_emergency_exit_preparation_does_not_close_the_window(window) -> None:
    window.navigate("recovery")
    page = window.page("recovery")
    window.context.paranoia.record_activity("browser_open", points=45)

    page.prepare_emergency_exit()

    assert window.isVisible()
    assert window.context.paranoia.restrictions_released()
    assert window.context.monitor.enabled() is False


def test_all_pages_fit_the_minimum_window_at_maximum_and_quarantine(window) -> None:
    context = window.context
    context.paranoia.advance_to_maximum_paranoia()
    context.demo.launch_application("calculator")
    context.paranoia.run_self_audit()
    context.paranoia.begin_quarantine()
    window.resize(1000, 650)
    QApplication.processEvents()

    for page_id in window.pages():
        window.navigate(page_id)
        QApplication.processEvents()
        page = window.page(page_id)
        content = page._scroll.widget()
        assert content.width() <= page._scroll.viewport().width() + 1, page_id


def test_header_badge_tracks_paranoia_state(window) -> None:
    window.context.paranoia.advance_to_maximum_paranoia()
    QApplication.processEvents()
    assert window.header.status_badge.text() == "MAXIMUM PARANOIA"
    assert window.header.status_badge.tone() == "danger"

    window.context.paranoia.quick_reset("badge test")
    QApplication.processEvents()
    assert window.header.status_badge.text() == "APPLICATION ONLINE"
