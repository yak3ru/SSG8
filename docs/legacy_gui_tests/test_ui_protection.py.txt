"""Offscreen integration checks for the protection interface."""

from __future__ import annotations

import pytest
from PySide6.QtWidgets import QApplication

from ssg8.core.context import build_context
from ssg8.storage.preferences import Preferences
from ssg8.storage.protection_store import ProtectionStore
from ssg8.ui import theme
from ssg8.ui.main_window import MainWindow


@pytest.fixture(scope="module")
def qapp():
    app = QApplication.instance() or QApplication([])
    theme.apply_theme(app)
    return app


@pytest.fixture
def window(qapp, tmp_path):
    context = build_context(
        Preferences(tmp_path / "preferences.json"),
        ProtectionStore(tmp_path / "data"),
    )
    window = MainWindow(context)
    window.show()
    qapp.processEvents()
    yield window
    window.close()
    qapp.processEvents()


def test_all_pages_navigate(window) -> None:
    assert len(window.pages()) == 14
    for page_id in window.pages():
        window.navigate(page_id)
        assert window.current_page_id == page_id


def test_pages_fit_the_minimum_window(window) -> None:
    window.resize(1000, 650)
    QApplication.processEvents()

    for page_id in window.pages():
        window.navigate(page_id)
        QApplication.processEvents()
        page = window.page(page_id)
        content = page._scroll.widget()
        assert content.width() <= page._scroll.viewport().width() + 1, page_id


def test_pages_fit_the_large_window(window) -> None:

def test_website_engine_flow(window) -> None:
    page = window.page("websites")
    assert page.add_entry("example.com", "block")[0]
    assert page.evaluate("https://example.com/login").action.value == "block"
    assert page.evaluate("https://unlisted.test/").action.value == "allow"
    assert page.visible_rules() == 1


def test_application_engine_flow(window) -> None:
    page = window.page("applications")
    assert page.add_entry("Calculator", r"C:\Windows\System32\calc.exe", "block")[0]
    decision = page.evaluate("Calculator", r"c:/windows/system32/calc.exe")
    assert decision.action.value == "block"


def test_download_engine_flow(window, tmp_path) -> None:
    page = window.page("downloads")
    sample = tmp_path / "sample.txt"
    sample.write_text("sample content", encoding="utf-8")

    decision = page.inspect_file(str(sample))

    assert decision.action.value == "allow"
    assert len(decision.metadata["sha256"]) == 64
    assert page.visible_inspections() == 1
    assert page._metadata_labels["name"].text() == "sample.txt"


def test_notification_flow(window) -> None:
    page = window.page("notifications")
    page._master_switch.setChecked(True)
    QApplication.processEvents()

    allowed = page.simulate("general", "Hello")
    assert allowed.action.value == "allow"

    page.context.state.set_notification_category_enabled("general", False)
    blocked = page.simulate("general", "Hello")
    assert blocked.action.value == "block"


def test_threat_monitor_and_logs_show_decisions(window) -> None:
    window.page("websites").evaluate("https://example.com/")
    window.page("applications").evaluate("Unknown")

    assert window.page("threat-monitor").visible_decisions() >= 2
    assert window.page("logs").visible_decisions() >= 2


def test_lockdown_page_reflects_module_states(window) -> None:
    page = window.page("lockdown")
    context = window.context
    context.protection.set_module_enabled("download", False)

    page.start_simulation("integration check")
    assert page.module_states()["download"] is True
    assert page.state_text() == "SIMULATED LOCKDOWN ACTIVE"

    page.end_simulation()
    assert page.module_states()["download"] is False
    assert page.state_text() == "INACTIVE"


def test_settings_policy_is_saved(window, tmp_path) -> None:
    page = window.page("settings")
    page._size_input.setText("48")
    page._extensions_input.setText(".exe, .ps1")
    page._save_protection_policy()

    settings = window.context.protection.settings()
    assert settings.max_file_size_mb == 48
    assert settings.blocked_extensions == (".exe", ".ps1")
    assert (tmp_path / "data" / "protection_settings.json").exists()


def test_dashboard_metrics_follow_real_decisions(window) -> None:
    dashboard = window.page("dashboard")
    window.page("websites").evaluate("https://example.com/")

    assert dashboard._evaluations_card.value() == "1"
    assert dashboard._allowed_card.value() == "1"
    assert dashboard._modules_card.value() == "4/4"
